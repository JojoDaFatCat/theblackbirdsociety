create table if not exists public.messages(id bigserial primary key, sender uuid not null references auth.users(id) on delete cascade, recipient uuid not null references auth.users(id) on delete cascade, body text not null check (char_length(body) between 1 and 1000), created_at timestamptz not null default now(), read_at timestamptz);
create index if not exists messages_pair on public.messages(least(sender,recipient), greatest(sender,recipient), created_at desc);
create index if not exists messages_rcpt on public.messages(recipient, read_at);
alter table public.messages enable row level security;
drop policy if exists "own messages read" on public.messages;
create policy "own messages read" on public.messages for select using (auth.uid() in (sender, recipient));
create table if not exists public.blocks(blocker uuid not null references auth.users(id) on delete cascade, blocked uuid not null references auth.users(id) on delete cascade, created_at timestamptz default now(), primary key(blocker, blocked));
alter table public.blocks enable row level security;
drop policy if exists "own blocks" on public.blocks;
create policy "own blocks" on public.blocks for select using (auth.uid() = blocker);
create table if not exists public.msg_reports(id bigserial primary key, reporter uuid references auth.users(id) on delete set null, reported uuid references auth.users(id) on delete set null, message_id bigint, body text, reason text, created_at timestamptz default now(), handled boolean default false);
alter table public.msg_reports enable row level security;

create or replace function public._uid_of(p_no bigint) returns uuid language sql stable security definer set search_path to 'public' as $$ select id from public.profiles where member_no = p_no $$;
create or replace function public._can_msg(a uuid, b uuid) returns boolean language sql stable security definer set search_path to 'public' as $$
  select a is not null and b is not null and a <> b
   and exists(select 1 from public.partners p where p.status='accepted' and ((p.requester=a and p.addressee=b) or (p.requester=b and p.addressee=a)))
   and not exists(select 1 from public.blocks k where (k.blocker=a and k.blocked=b) or (k.blocker=b and k.blocked=a)) $$;

create or replace function public.msg_send(p_no bigint, p_body text) returns text language plpgsql security definer set search_path to 'public' as $$
declare me uuid := auth.uid(); them uuid := public._uid_of(p_no); b text := btrim(coalesce(p_body,'')); nm text;
begin
  if me is null then return 'login'; end if;
  if b = '' or char_length(b) > 1000 then return 'bad'; end if;
  if not public._can_msg(me, them) then return 'no'; end if;
  if (select count(*) from public.messages where sender = me and created_at > now() - interval '1 minute') >= 15 then return 'slow'; end if;
  insert into public.messages(sender, recipient, body) values (me, them, b);
  if not exists(select 1 from public.notifications n where n.user_id = them and n.kind = 'msg' and not n.read and (n.data->>'from')::bigint = (select member_no from public.profiles where id = me)) then
    select coalesce(nullif(username,''), split_part(name,' ',1)) into nm from public.profiles where id = me;
    perform public._notify(them, 'msg', jsonb_build_object('name', nm, 'from', (select member_no from public.profiles where id = me)), false);
  end if;
  return 'ok';
end $$;

create or replace function public.msg_thread(p_no bigint) returns table(id bigint, mine boolean, body text, created_at timestamptz, read boolean) language plpgsql security definer set search_path to 'public' as $$
declare me uuid := auth.uid(); them uuid := public._uid_of(p_no);
begin
  if me is null or them is null then return; end if;
  update public.messages m set read_at = now() where m.recipient = me and m.sender = them and m.read_at is null;
  update public.notifications n set read = true where n.user_id = me and n.kind = 'msg' and not n.read and (n.data->>'from')::bigint = p_no;
  return query select x.id, x.mine, x.body, x.created_at, x.read from (select m.id, m.sender = me as mine, m.body, m.created_at, m.read_at is not null as read from public.messages m where (m.sender = me and m.recipient = them) or (m.sender = them and m.recipient = me) order by m.created_at desc limit 200) x order by x.created_at;
end $$;

create or replace function public.msg_inbox() returns table(member_no bigint, name text, avatar text, last_body text, last_at timestamptz, last_mine boolean, unread bigint, can boolean) language sql stable security definer set search_path to 'public' as $$
  with me as (select auth.uid() u), c as (
    select case when m.sender = (select u from me) then m.recipient else m.sender end other, m.* from public.messages m where (select u from me) in (m.sender, m.recipient)),
  l as (select distinct on (other) other, body, created_at, sender = (select u from me) mine from c order by other, created_at desc)
  select p.member_no, coalesce(nullif(p.username,''), split_part(p.name,' ',1)), p.avatar_url, l.body, l.created_at, l.mine,
    (select count(*) from public.messages x where x.sender = l.other and x.recipient = (select u from me) and x.read_at is null),
    public._can_msg((select u from me), l.other)
  from l join public.profiles p on p.id = l.other order by l.created_at desc $$;

create or replace function public.msg_unread() returns bigint language sql stable security definer set search_path to 'public' as $$
  select count(*) from public.messages where recipient = auth.uid() and read_at is null $$;

create or replace function public.msg_block(p_no bigint, p_on boolean default true) returns text language plpgsql security definer set search_path to 'public' as $$
declare me uuid := auth.uid(); them uuid := public._uid_of(p_no);
begin if me is null or them is null or me = them then return 'bad'; end if;
  if p_on then insert into public.blocks(blocker, blocked) values (me, them) on conflict do nothing; else delete from public.blocks where blocker = me and blocked = them; end if;
  return 'ok'; end $$;

create or replace function public.msg_report(p_no bigint, p_msg bigint, p_reason text) returns text language plpgsql security definer set search_path to 'public' as $$
declare me uuid := auth.uid(); them uuid := public._uid_of(p_no); b text; owner uuid;
begin if me is null or them is null then return 'bad'; end if;
  select body into b from public.messages where id = p_msg and sender = them and recipient = me;
  insert into public.msg_reports(reporter, reported, message_id, body, reason) values (me, them, p_msg, b, left(coalesce(p_reason,''),300));
  select id into owner from public.profiles where member_no = 1;
  perform public._notify(owner, 'msg_report', jsonb_build_object('name', (select coalesce(nullif(username,''),split_part(name,' ',1)) from public.profiles where id = them)), false);
  return 'ok'; end $$;

create or replace function public.admin_msg_reports() returns table(id bigint, reporter text, reported text, reported_no bigint, body text, reason text, created_at timestamptz, handled boolean) language sql stable security definer set search_path to 'public' as $$
  select r.id, a.username, b.username, b.member_no, r.body, r.reason, r.created_at, r.handled from public.msg_reports r left join public.profiles a on a.id = r.reporter left join public.profiles b on b.id = r.reported where public._is_owner() order by r.created_at desc limit 50 $$;
create or replace function public.admin_report_done(p_id bigint) returns void language sql security definer set search_path to 'public' as $$ update public.msg_reports set handled = true where id = p_id and public._is_owner() $$;

revoke all on function public._uid_of(bigint), public._can_msg(uuid,uuid) from public, anon, authenticated;
grant execute on function public.msg_send(bigint,text), public.msg_thread(bigint), public.msg_inbox(), public.msg_unread(), public.msg_block(bigint,boolean), public.msg_report(bigint,bigint,text), public.admin_msg_reports(), public.admin_report_done(bigint) to authenticated;
select 'done';
